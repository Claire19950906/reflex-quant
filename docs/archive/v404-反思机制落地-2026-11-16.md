```sql
-- 两表同决策的出场价是否还分叉（应为 0）
SELECT COUNT(*) FROM trade_reflections_v2 r
  JOIN quick_sim_trades q ON q.decision_id = r.decision_id
 WHERE q.settled = 1 AND q.pnl_pct IS NOT NULL
   AND (r.pnl_pct IS NULL OR ABS(r.pnl_pct - q.pnl_pct) > 0.5);

-- 假胜是否归零
SELECT COALESCE(SUM(hit),0) AS sum_hit FROM trade_reflections_v2;
```
